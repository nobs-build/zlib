unit zlib

build: shared static example minigzip examplestatic minigzipstatic @PLATFORM_BUILD_TARGETS
{
    NO_TOOL
}

generate_zconf: zconf_generator_tool
{
    GENERATE_ZCONF
    ZCONF_TOOL = "generate"
    GENERATOR = zconf_generator_tool.BINARY
    SOURCE_ZCONF = P"(zconf.h)"
    GENERATED_INCLUDE = P"(out/generated/${PROFILE})"
    OUTPUT_ZCONF = P"(out/generated/${PROFILE}/zconf.h)"
}

zconf_generator_tool:
{
    C_FILES = P"(tools/gen_zconf.c)"
    OUTPUT_NAME = "gen_zconf${HOST_EXECUTABLE_EXTENSION}"
}

zlib_sources:
{
    NO_TOOL
    ^C_FILES = [
        P"(adler32.c)",
        P"(compress.c)",
        P"(crc32.c)",
        P"(deflate.c)",
        P"(gzclose.c)",
        P"(gzlib.c)",
        P"(gzread.c)",
        P"(gzwrite.c)",
        P"(infback.c)",
        P"(inflate.c)",
        P"(inftrees.c)",
        P"(inffast.c)",
        P"(trees.c)",
        P"(uncompr.c)",
        P"(zutil.c)"
    ]
}

shared: generate_zconf +core(core_config) +@SHARED_EXTRAS
{
    SHARED
    OBJS
    LINK_FLAGS = []
    LINK_FLAGS += SHARED_LINK_FLAGS
    OUTPUT_NAME = "libz${SHARED_LIBRARY_EXTENSION}"
    core_config =
    {
        SHARED_BUILD
        C_FLAGS = SHARED_C_FLAGS
        C_DEFINES = SHARED_C_DEFINES
    }
}

static: generate_zconf +core(core_config)
{
    OBJS
    STATIC
    OUTPUT_NAME = "libz${STATIC_LIBRARY_EXTENSION}"
    core_config =
    {
        C_FLAGS = []
        C_DEFINES = []
    }
}

core(config): +zlib_sources
{
    C_FILES = []
    C_FLAGS = []
    C_FLAGS += [C_COMPILE_FLAG]
    C_FLAGS += config.C_FLAGS
    C_DEFINES += ["ZLIB_BUILD"]
    C_DEFINES += PLATFORM_C_DEFINES
    C_DEFINES += config.C_DEFINES
    INCLUDE = [P"(out/generated/${PROFILE})", P"(.)"]
}

windows_resource: +windows_resource_objects
{
    NO_TOOL
}

windows_resource_objects: windows_resource_compile
{
    NO_TOOL
    ^OBJS = windows_resource_compile.RES
}

windows_resource_compile:
{
    RC_FILE = P"(win32/zlib-cmake.rc)"
    OUTPUT_NAME = "zlib.res"
}
