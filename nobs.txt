unit zlib

build: @BUILD_TARGETS
{
    NO_TOOL
}

generate_zconf: zconf_generator_tool
{
    GENERATE_ZCONF
    ZCONF_TOOL = "generate"
    GENERATOR = zconf_generator_tool.BINARY
    SOURCE_ZCONF = P"(zconf.h)"
    GENERATED_INCLUDE = P"(out/generated/${PROFILE})"
    OUTPUT_ZCONF = P"(out/generated/${PROFILE}/zconf.h)"
}

zconf_generator_tool:
{
    C_FILES = P"(tools/gen_zconf.c)"
    OUTPUT_NAME = "gen_zconf"

    if OS_FAMILY = "windows"
    {
        OUTPUT_NAME = "gen_zconf.exe"
    }
}

zlib_sources:
{
    NO_TOOL
    ^C_FILES = [
        P"(adler32.c)",
        P"(compress.c)",
        P"(crc32.c)",
        P"(deflate.c)",
        P"(gzclose.c)",
        P"(gzlib.c)",
        P"(gzread.c)",
        P"(gzwrite.c)",
        P"(infback.c)",
        P"(inflate.c)",
        P"(inftrees.c)",
        P"(inffast.c)",
        P"(trees.c)",
        P"(uncompr.c)",
        P"(zutil.c)"
    ]
}
